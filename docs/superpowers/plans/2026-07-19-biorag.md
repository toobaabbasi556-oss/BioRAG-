# BioRAG Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and measure a retrieval-augmented QA system over ~3,000 PubMed musculoskeletal-biomechanics abstracts, where a fine-tuned embedding model is compared against off-the-shelf embeddings and a BM25 baseline on a hand-labeled question set.

**Architecture:** Six modules with one-directional data flow — `ingest` writes a corpus to disk, `index` builds FAISS and BM25 indexes over it, `retrieve` exposes one `Retriever` interface with three interchangeable implementations, `train` fine-tunes the embedding model, `generate` calls an LLM over retrieved context, and `eval` runs all retrievers through identical code to produce the ablation table. Each stage persists its output, so a failure downstream never forces re-running an upstream stage.

**Tech Stack:** Python 3.12, sentence-transformers, FAISS (`faiss-cpu`), `rank_bm25`, Groq API (free tier), Streamlit, pytest, Docker.

**Spec:** `docs/superpowers/specs/2026-07-19-biorag-design.md`

```python
from dataclasses import dataclass, asdict


@dataclass(frozen=True)
class Chunk:
    """One retrievable unit of text. For this corpus, one chunk == one abstract."""

    id: str  # unique chunk id, e.g. "12345678" or "12345678-1" when split
    pmid: str
    title: str
    text: str
    journal: str
    year: str

    def to_dict(self) -> dict:
        return asdict(self)

    @staticmethod
    def from_dict(d: dict) -> "Chunk":
        return Chunk(**d)
```

```python
# src/biorag/ingest/clean.py
"""Cleaning and chunking of raw PubMed abstract text."""

import re

from biorag.models import Chunk

# Publisher boilerplate carries no retrievable information and pollutes embeddings.
_BOILERPLATE = re.compile(
    r"(copyright|©)\s*\d{4}.*$|all rights reserved\.?\s*$",
    re.IGNORECASE | re.DOTALL,
)


def clean_text(raw: str) -> str:
    """Collapse whitespace and strip publisher boilerplate."""
    text = _BOILERPLATE.sub("", raw)
    return re.sub(r"\s+", " ", text).strip()


def _split_sentences(text: str) -> list[str]:
    parts = re.split(r"(?<=[.!?])\s+", text.strip())
    return [p for p in parts if p]


def chunk_abstract(
    pmid: str,
    title: str,
    text: str,
    journal: str,
    year: str,
    max_words: int = 220,
) -> list[Chunk]:
    """Return one Chunk per abstract, splitting only if it exceeds max_words.

    Splits on sentence boundaries with a one-sentence overlap so a fact spanning
    a boundary is still retrievable from at least one chunk.
    """
    text = clean_text(text)

    if len(text.split()) <= max_words:
        return [Chunk(id=pmid, pmid=pmid, title=title, text=text,
                      journal=journal, year=year)]

    sentences = _split_sentences(text)
    chunks: list[Chunk] = []
    current: list[str] = []

    for sentence in sentences:
        current.append(sentence)
        if sum(len(s.split()) for s in current) >= max_words:
            chunks.append(" ".join(current))
            current = [current[-1]]  # one-sentence overlap

    remaining = [s for s in current if s]
    # Avoid emitting a trailing chunk that is only the overlap sentence.
    if remaining and (len(remaining) > 1 or not chunks):
        chunks.append(" ".join(remaining))

    return [
        Chunk(id=f"{pmid}-{i}", pmid=pmid, title=title, text=body,
              journal=journal, year=year)
        for i, body in enumerate(chunks)
    ]
```

```python
# src/biorag/ingest/fetch.py
"""Resumable PubMed fetching via Entrez E-utilities."""

import json
import time
from pathlib import Path

from Bio import Entrez

# NCBI requires a contact email with every request.
Entrez.email = "Toobaabbasi556@gmail.com"

SEARCH_QUERY = (
    '("spinal load"[All Fields] OR "spine biomechanics"[All Fields] '
    'OR "gait analysis"[All Fields] OR "musculoskeletal model"[All Fields] '
    'OR "lumbar spine"[All Fields] OR "joint kinematics"[All Fields] '
    'OR "motion capture"[All Fields]) AND hasabstract'
)


def search_pmids(query: str = SEARCH_QUERY, retmax: int = 3000) -> list[str]:
    """Return up to retmax PMIDs matching the query."""
    handle = Entrez.esearch(db="pubmed", term=query, retmax=retmax)
    record = Entrez.read(handle)
    handle.close()
    return list(record["IdList"])


def _extract(article: dict) -> dict | None:
    """Pull the fields we need out of one Entrez article record."""
    try:
        citation = article["MedlineCitation"]
        art = citation["Article"]
        abstract_parts = art["Abstract"]["AbstractText"]
        # AbstractText entries are str subclasses; structured abstracts have several.
        text = " ".join(str(p) for p in abstract_parts).strip()
        if not text:
            return None
        pub_date = (
            art.get("Journal", {}).get("JournalIssue", {}).get("PubDate", {})
        )
        # PubDate carries either a Year or only a MedlineDate string like "2019 Jul-Aug".
        year = str(pub_date.get("Year") or pub_date.get("MedlineDate", "unknown"))[:4]
        return {
            "pmid": str(citation["PMID"]),
            "title": str(art.get("ArticleTitle", "")),
            "text": text,
            "journal": str(art.get("Journal", {}).get("Title", "unknown")),
            "year": year,
        }
    except (KeyError, IndexError):
        # Records without abstracts or with unexpected shape are skipped, not fatal.
        return None


def fetch_records(pmids: list[str], out_path: Path, batch_size: int = 100) -> int:
    """Fetch abstracts for pmids, appending JSON lines to out_path.

    Resumable: PMIDs already present in out_path are skipped, so an interrupted
    run continues rather than restarting.
    """
    out_path.parent.mkdir(parents=True, exist_ok=True)
    seen: set[str] = set()
    if out_path.exists():
        with out_path.open() as f:
            for line in f:
                seen.add(json.loads(line)["pmid"])

    todo = [p for p in pmids if p not in seen]
    print(f"{len(seen)} already fetched, {len(todo)} to go")

    written = 0
    with out_path.open("a") as f:
        for i in range(0, len(todo), batch_size):
            batch = todo[i : i + batch_size]
            for attempt in range(3):
                try:
                    handle = Entrez.efetch(
                        db="pubmed", id=",".join(batch), retmode="xml"
                    )
                    records = Entrez.read(handle)
                    handle.close()
                    break
                except Exception as e:  # network/rate-limit failures are transient
                    if attempt == 2:
                        raise
                    print(f"  batch failed ({e}), retrying in {2 ** attempt}s")
                    time.sleep(2**attempt)

            for article in records.get("PubmedArticle", []):
                rec = _extract(article)
                if rec:
                    f.write(json.dumps(rec) + "\n")
                    written += 1
            f.flush()
            print(f"  fetched {i + len(batch)}/{len(todo)}")
            time.sleep(0.4)  # NCBI allows 3 req/s without an API key

    return written
```

```python
# src/biorag/ingest/build_corpus.py
"""CLI: fetch PubMed records, clean and chunk them into data/corpus.jsonl.

Usage:  python -m biorag.ingest.build_corpus --limit 3000
"""

import argparse
import json
from pathlib import Path

from biorag.ingest.clean import chunk_abstract
from biorag.ingest.fetch import fetch_records, search_pmids

RAW_PATH = Path("data/raw.jsonl")
CORPUS_PATH = Path("data/corpus.jsonl")


def build(limit: int) -> None:
    pmids = search_pmids(retmax=limit)
    print(f"search returned {len(pmids)} pmids")
    fetch_records(pmids, RAW_PATH)

    chunks = []
    with RAW_PATH.open() as f:
        for line in f:
            rec = json.loads(line)
            chunks.extend(
                chunk_abstract(
                    pmid=rec["pmid"], title=rec["title"], text=rec["text"],
                    journal=rec["journal"], year=rec["year"],
                )
            )

    with CORPUS_PATH.open("w") as f:
        for chunk in chunks:
            f.write(json.dumps(chunk.to_dict()) + "\n")

    print(f"wrote {len(chunks)} chunks to {CORPUS_PATH}")


def load_corpus(path: Path = CORPUS_PATH) -> list:
    """Load the corpus as Chunk objects. Used by every downstream module."""
    from biorag.models import Chunk

    with path.open() as f:
        return [Chunk.from_dict(json.loads(line)) for line in f]


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--limit", type=int, default=3000)
    build(parser.parse_args().limit)
```

```python
# src/biorag/retrieve/bm25.py
"""Lexical baseline. This is the number the fine-tuned model must beat."""

import re

from rank_bm25 import BM25Okapi

from biorag.models import Chunk


def _tokenize(text: str) -> list[str]:
    return re.findall(r"[a-z0-9]+", text.lower())


class BM25Retriever:
    def __init__(self, chunks: list[Chunk]) -> None:
        self._chunks = chunks
        # Title carries strong signal in abstracts, so index it alongside the body.
        corpus = [_tokenize(f"{c.title} {c.text}") for c in chunks]
        self._bm25 = BM25Okapi(corpus)

    def search(self, query: str, k: int = 5) -> list[Chunk]:
        scores = self._bm25.get_scores(_tokenize(query))
        ranked = sorted(range(len(scores)), key=lambda i: scores[i], reverse=True)
        return [self._chunks[i] for i in ranked[:k]]
```

```python
# src/biorag/index/dense.py
"""Embedding and FAISS index construction."""

from pathlib import Path

import faiss
import numpy as np
from sentence_transformers import SentenceTransformer

from biorag.models import Chunk

BASE_MODEL = "sentence-transformers/all-MiniLM-L6-v2"


def embed(model: SentenceTransformer, texts: list[str]) -> np.ndarray:
    """Embed texts and L2-normalise, so inner product == cosine similarity."""
    vectors = model.encode(
        texts, batch_size=64, show_progress_bar=True, convert_to_numpy=True
    )
    faiss.normalize_L2(vectors)
    return vectors


def build_index(chunks: list[Chunk], model_name: str = BASE_MODEL) -> faiss.Index:
    """Build an exact inner-product index. At ~3k chunks, exact search is fast
    and avoids tuning an approximate index for no measurable benefit."""
    model = SentenceTransformer(model_name)
    texts = [f"{c.title}. {c.text}" for c in chunks]
    vectors = embed(model, texts)
    index = faiss.IndexFlatIP(vectors.shape[1])
    index.add(vectors)
    return index


def save_index(index: faiss.Index, path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    faiss.write_index(index, str(path))


def load_index(path: Path) -> faiss.Index:
    return faiss.read_index(str(path))
```

```python
# src/biorag/retrieve/dense.py
"""Embedding-based retriever.

`model_name` is a constructor argument, which is the whole point: the off-the-shelf
run and the fine-tuned run are the same class with a different path.
"""

import faiss
from sentence_transformers import SentenceTransformer

from biorag.index.dense import BASE_MODEL, build_index
from biorag.models import Chunk


class DenseRetriever:
    def __init__(
        self,
        chunks: list[Chunk],
        model_name: str = BASE_MODEL,
        index: faiss.Index | None = None,
    ) -> None:
        self._chunks = chunks
        self._model = SentenceTransformer(model_name)
        self._index = index if index is not None else build_index(chunks, model_name)

    def search(self, query: str, k: int = 5) -> list[Chunk]:
        vector = self._model.encode([query], convert_to_numpy=True)
        faiss.normalize_L2(vector)
        _scores, ids = self._index.search(vector, k)
        return [self._chunks[i] for i in ids[0] if i != -1]
```

```python
"""Thin Groq wrapper. Fails loudly rather than returning empty answers."""

import os
import time

from dotenv import load_dotenv
from groq import Groq

load_dotenv()

MODEL = "llama-3.3-70b-versatile"


class LLMError(RuntimeError):
    pass


def complete(prompt: str, temperature: float = 0.0, max_tokens: int = 700) -> str:
    """Send a prompt, return the completion text. One retry on transient failure."""
    key = os.environ.get("GROQ_API_KEY")
    if not key:
        raise LLMError("GROQ_API_KEY is not set — copy .env.example to .env")

    client = Groq(api_key=key)
    for attempt in range(2):
        try:
            response = client.chat.completions.create(
                model=MODEL,
                messages=[{"role": "user", "content": prompt}],
                temperature=temperature,
                max_tokens=max_tokens,
            )
            return response.choices[0].message.content or ""
        except Exception as e:
            if attempt == 1:
                raise LLMError(f"Groq call failed after retry: {e}") from e
            time.sleep(2)
    raise LLMError("unreachable")
```

Temperature is 0.0 because the evaluation must be reproducible — a nondeterministic

```python
# src/biorag/generate/answer.py
"""Prompt assembly and grounded answer generation."""

from dataclasses import dataclass

from biorag.generate.llm import complete
from biorag.models import Chunk
from biorag.retrieve.base import Retriever

NOT_FOUND = "Not found in corpus."

PROMPT = """You are answering questions using only the provided biomedical abstracts.

Rules:
- Use ONLY the abstracts below. Do not use outside knowledge.
- Cite the PMID in square brackets after each claim, e.g. [12345678].
- If the abstracts do not answer the question, reply exactly: {not_found}

Abstracts:
{context}

Question: {question}

Answer:"""


@dataclass
class Answer:
    text: str
    chunks: list[Chunk]


def _format_context(chunks: list[Chunk]) -> str:
    return "\n\n".join(
        f"[{c.pmid}] {c.title}\n{c.text}" for c in chunks
    )


def answer_question(question: str, retriever: Retriever, k: int = 5) -> Answer:
    """Retrieve, then generate a grounded answer with citations."""
    chunks = retriever.search(question, k=k)
    if not chunks:
        return Answer(text=NOT_FOUND, chunks=[])

    prompt = PROMPT.format(
        not_found=NOT_FOUND, context=_format_context(chunks), question=question
    )
    return Answer(text=complete(prompt).strip(), chunks=chunks)
```

```python
# src/biorag/evaluate/metrics.py
"""Retrieval metrics. Pure functions, no I/O — so they can be tested exactly."""


def recall_at_k(retrieved: list[str], relevant: set[str], k: int) -> float:
    """Fraction of relevant documents that appear in the top k retrieved."""
    if not relevant:
        raise ValueError("recall_at_k requires a non-empty relevant set")
    top = set(retrieved[:k])
    return len(top & relevant) / len(relevant)


def mrr(retrieved: list[str], relevant: set[str]) -> float:
    """Reciprocal rank of the first relevant document; 0.0 if none retrieved."""
    for position, doc_id in enumerate(retrieved, start=1):
        if doc_id in relevant:
            return 1.0 / position
    return 0.0
```

```python
# src/biorag/train/pairs.py
"""Weakly-supervised training pair generation for embedding fine-tuning.

MultipleNegativesRankingLoss needs only positive pairs — negatives come from
other examples in the batch — so the pairs here are all (query, positive) shaped.
"""

import json
from pathlib import Path

from biorag.models import Chunk

QUESTIONS_PATH = Path("eval/questions.jsonl")


def load_questions(path: Path = QUESTIONS_PATH) -> list[dict]:
    with path.open() as f:
        return [json.loads(line) for line in f]


def held_out_pmids(questions: list[dict]) -> set[str]:
    """Every PMID cited by the eval set. These must never appear in training."""
    return {pmid for q in questions for pmid in q["relevant_pmids"]}


def build_title_pairs(
    chunks: list[Chunk], held_out: set[str]
) -> list[tuple[str, str]]:
    """(title, body) pairs — a title is a natural short query for its abstract."""
    return [
        (c.title, c.text)
        for c in chunks
        if c.pmid not in held_out and c.text.strip() and c.title.strip()
    ]
```

```python
Usage:  PYTHONPATH=src python -m biorag.train.finetune --epochs 1

import argparse

from sentence_transformers import InputExample, SentenceTransformer, losses

from biorag.ingest.build_corpus import load_corpus

def main(epochs: int, batch_size: int) -> None:
    chunks = load_corpus()
    held = held_out_pmids(load_questions())
    pairs = build_title_pairs(chunks, held_out=held)

    model = SentenceTransformer(BASE_MODEL)
    examples = [InputExample(texts=[q, p]) for q, p in pairs]
    loader = DataLoader(examples, shuffle=True, batch_size=batch_size)
    loss = losses.MultipleNegativesRankingLoss(model)

    model.fit(
        train_objectives=[(loader, loss)],
        epochs=epochs,
        warmup_steps=int(0.1 * len(loader) * epochs),
        show_progress_bar=True,
    )
    model.save(OUT_DIR)
    print(f"saved to {OUT_DIR}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--epochs", type=int, default=1)
    parser.add_argument("--batch-size", type=int, default=32)
    main(*vars(parser.parse_args()).values())
```

```python
# src/biorag/evaluate/run_retrieval.py
"""CLI: evaluate all three retrievers on the question set.

Usage:  PYTHONPATH=src python -m biorag.evaluate.run_retrieval
Writes: results/retrieval.md
"""

from pathlib import Path
from statistics import mean

from biorag.evaluate.metrics import mrr, recall_at_k
from biorag.index.dense import load_index
from biorag.ingest.build_corpus import load_corpus
from biorag.retrieve.bm25 import BM25Retriever
from biorag.retrieve.dense import DenseRetriever
from biorag.train.pairs import load_questions

OUT_PATH = Path("results/retrieval.md")


def _distinct_pmids(chunks: list, limit: int) -> list[str]:
    """Collapse chunks to distinct PMIDs, best-ranked first.

    ~2/3 of abstracts split into multiple chunks, so a top-10 chunk list holds
    only ~9 distinct PMIDs. Ground truth is labeled by PMID, so without this,
    "recall@10" would silently mean a different k on every query.
    """
    seen: list[str] = []
    for chunk in chunks:
        if chunk.pmid not in seen:
            seen.append(chunk.pmid)
        if len(seen) == limit:
            break
    return seen


def evaluate(retriever, questions: list[dict]) -> dict[str, float]:
    """Score one retriever. Unanswerable questions are excluded — recall is
    undefined without a relevant set, and they are scored separately by hand."""
    scored = [q for q in questions if q["relevant_pmids"]]
    r5, r10, rr = [], [], []

    for q in scored:
        # Over-fetch chunks so that 10 DISTINCT pmids are reliably available.
        hits = retriever.search(q["question"], k=30)
        retrieved = _distinct_pmids(hits, limit=10)
        relevant = set(q["relevant_pmids"])
        r5.append(recall_at_k(retrieved, relevant, k=5))
        r10.append(recall_at_k(retrieved, relevant, k=10))
        rr.append(mrr(retrieved, relevant))

    return {"recall@5": mean(r5), "recall@10": mean(r10), "MRR": mean(rr)}


def main() -> None:
    chunks = load_corpus()
    questions = load_questions()

    retrievers = {
        "BM25 (baseline)": BM25Retriever(chunks),
        "MiniLM (off-the-shelf)": DenseRetriever(
            chunks, index=load_index(Path("data/base.faiss"))
        ),
        "MiniLM (fine-tuned)": DenseRetriever(
            chunks,
            model_name="models/finetuned-minilm",
            index=load_index(Path("data/finetuned.faiss")),
        ),
    }

    rows = {name: evaluate(r, questions) for name, r in retrievers.items()}

    lines = [
        f"# Retrieval results\n",
        f"Corpus: {len(chunks)} chunks. "
        f"Scored on {len([q for q in questions if q['relevant_pmids']])} "
        f"answerable questions.\n",
        "| Retriever | recall@5 | recall@10 | MRR |",
        "|---|---|---|---|",
    ]
    for name, m in rows.items():
        lines.append(
            f"| {name} | {m['recall@5']:.3f} | {m['recall@10']:.3f} | {m['MRR']:.3f} |"
        )

    OUT_PATH.parent.mkdir(exist_ok=True)
    OUT_PATH.write_text("\n".join(lines) + "\n")
    print("\n".join(lines))


if __name__ == "__main__":
    main()
```

```python
"""Minimal Streamlit UI. Run: PYTHONPATH=src streamlit run src/biorag/app.py"""

from pathlib import Path

import streamlit as st

from biorag.generate.answer import answer_question
from biorag.index.dense import load_index
from biorag.ingest.build_corpus import load_corpus
from biorag.retrieve.dense import DenseRetriever

MODEL_PATH = "models/finetuned-minilm"
INDEX_PATH = Path("data/finetuned.faiss")


@st.cache_resource
def get_retriever() -> DenseRetriever:
    """Cached so the model and index load once, not on every keystroke."""
    chunks = load_corpus()
    return DenseRetriever(chunks, model_name=MODEL_PATH, index=load_index(INDEX_PATH))


st.title("BioRAG")
st.caption("Question answering over musculoskeletal biomechanics abstracts")

question = st.text_input("Question", placeholder="How does lifting posture affect spinal load?")

if question:
    with st.spinner("Retrieving and generating…"):
        result = answer_question(question, get_retriever())

    st.markdown(result.text)
    st.divider()
    st.subheader("Sources")
    for chunk in result.chunks:
        with st.expander(f"[{chunk.pmid}] {chunk.title}"):
            st.write(chunk.text)
            st.caption(f"{chunk.journal}, {chunk.year}")
```

```dockerfile
ENV PYTHONPATH=/app/src
EXPOSE 8501

CMD ["streamlit", "run", "src/biorag/app.py", \
     "--server.address=0.0.0.0", "--server.port=8501"]
```
